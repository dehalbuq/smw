cmake_minimum_required(VERSION 3.22.1)
project(SMWAndroidTV C CXX)
if(NOT ANDROID)
  message(FATAL_ERROR "Android NDK required")
endif()
if(NOT EXISTS "${SMW_ANDROID_SDL2}/CMakeLists.txt")
  message(FATAL_ERROR "SMW_ANDROID_SDL2 must point at pinned SDL2 sources")
endif()

set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${SMW_ANDROID_SDL2}" "${CMAKE_BINARY_DIR}/android-sdl2")
if(SMW_ANDROID_STUB)
  # ROM-FREE infrastructure test; this is deliberately NOT a playable game.
  add_library(main SHARED android_stub.c)
  target_link_libraries(main PRIVATE SDL2::SDL2 log android)
  message(STATUS "SMW_ANDROID_STUB=ON: TV shell only, no game engine")
  return()
endif()
if(NOT EXISTS "${SMW_ANDROID_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "SMW_ANDROID_SOURCE must point at prepared game checkout")
endif()
add_subdirectory("${SMW_ANDROID_SOURCE}" "${CMAKE_BINARY_DIR}/android-smw")

if(NOT TARGET SuperMarioWorldSNESRecomp)
  message(FATAL_ERROR "Missing SMW native game library")
endif()
set_target_properties(SuperMarioWorldSNESRecomp PROPERTIES OUTPUT_NAME main)
target_link_libraries(SuperMarioWorldSNESRecomp PRIVATE log android EGL GLESv2)
get_target_property(SMW_TGT_TYPE SuperMarioWorldSNESRecomp TYPE)
if(NOT SMW_TGT_TYPE STREQUAL "SHARED_LIBRARY")
  message(FATAL_ERROR "Android target is not a native shared library")
endif()
